CREATE TABLE notifications (
    id INT AUTO_INCREMENT PRIMARY KEY,
    type VARCHAR(50) NOT NULL, -- 'vendor_request', 'product_moderation', 'order_high', 'order_problem', 'support_message', 'security_alert'
    title VARCHAR(255) NOT NULL,
    message TEXT NOT NULL,
    lien VARCHAR(500) NULL,
    icone VARCHAR(50) DEFAULT 'fa-bell',
    couleur VARCHAR(20) DEFAULT 'blue',
    est_lu TINYINT(1) DEFAULT 0,
    priorite ENUM('low', 'medium', 'high') DEFAULT 'medium',
    source_type VARCHAR(50) NULL, -- 'user', 'product', 'order', 'system'
    source_id INT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_type (type),
    INDEX idx_est_lu (est_lu),
    INDEX idx_created_at (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE notification_settings (

    type VARCHAR(50) NOT NULL UNIQUE,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

-- Table des templates d'emails
CREATE TABLE IF NOT EXISTS email_templates (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nom VARCHAR(255) NOT NULL,
    slug VARCHAR(255) NOT NULL UNIQUE,
    declencheur VARCHAR(255) NOT NULL,
    objet VARCHAR(255) NOT NULL,
    contenu TEXT NOT NULL,
    bouton_texte VARCHAR(100) DEFAULT 'Voir les offres',
    bouton_url VARCHAR(500) DEFAULT '#',
    bg_color VARCHAR(50) DEFAULT 'bg-amber-100',
    text_color VARCHAR(50) DEFAULT 'text-amber-600',
    icon VARCHAR(50) DEFAULT 'fa-envelope-open-text',
    statut ENUM('active', 'inactive') DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_statut (statut)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Table des logs d'emails
CREATE TABLE IF NOT EXISTS email_logs (
    id INT AUTO_INCREMENT PRIMARY KEY,
    template_id INT NULL,
    destinataire VARCHAR(255) NOT NULL,
    sujet VARCHAR(255) NOT NULL,
    ouvert TINYINT(1) DEFAULT 0,
    clic TINYINT(1) DEFAULT 0,
    date_envoi TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_destinataire (destinataire),
    INDEX idx_date (date_envoi),
    FOREIGN KEY (template_id) REFERENCES email_templates(id) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

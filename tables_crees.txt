CREATE TABLE `bannieres` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `titre` varchar(255) NOT NULL,
  `sous_titre` varchar(255) DEFAULT NULL,
  `image` varchar(255) NOT NULL,
  `texte_bouton` varchar(100) DEFAULT 'Voir les offres',
  `url_destination` varchar(500) NOT NULL,
  `date_debut` datetime NOT NULL,
  `date_fin` datetime NOT NULL,
  `statut` enum('active','inactive') DEFAULT 'active',
  `ordre_affichage` int(11) DEFAULT 0,
  `vues` int(11) DEFAULT 0,
  `clics` int(11) DEFAULT 0,
  `created_at` timestamp NULL DEFAULT current_timestamp(),
  `updated_at` timestamp NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;CREATE TABLE `codes_promo` (

  `type` enum('percentage','fixed') DEFAULT 'percentage',

  `montant_minimum` decimal(10,2) DEFAULT 0,
  `utilisations_max` int(11) DEFAULT NULL,
  `utilisations_actuelles` int(11) DEFAULT 0,
  `date_expiration` datetime NOT NULL,
  `categorie_id` int(11) DEFAULT NULL,
  `produit_id` int(11) DEFAULT NULL,
  `utilisateur_id` int(11) DEFAULT NULL,
  `statut` enum('active','inactive','expire') DEFAULT 'active',
  `created_at` timestamp NULL DEFAULT current_timestamp(),
  `updated_at` timestamp NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  PRIMARY KEY (`id`),
  UNIQUE KEY `code` (`code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;CREATE TABLE `promo_utilisations` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `code_promo_id` int(11) NOT NULL,
  `commande_id` int(11) NOT NULL,
  `utilisateur_id` int(11) NOT NULL,
  `montant_initial` decimal(10,2) NOT NULL,
  `remise` decimal(10,2) NOT NULL,
  `montant_final` decimal(10,2) NOT NULL,
  `date_utilisation` timestamp NULL DEFAULT current_timestamp(),
  PRIMARY KEY (`id`),
  KEY `code_promo_id` (`code_promo_id`),

  KEY `utilisateur_id` (`utilisateur_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;CREATE TABLE `banniere_stats` (
  `id` int(11) NOT NULL AUTO_INCREMENT,

  `date` date NOT NULL,
  `vues` int(11) DEFAULT 0,
  `clics` int(11) DEFAULT 0,
  PRIMARY KEY (`id`),
  UNIQUE KEY `banniere_date` (`banniere_id`,`date`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

SELECT 
    p.id,
    CASE WHEN FLOOR(RAND() * 3) = 0 THEN FLOOR(1 + RAND() * 50) ELSE NULL END as user_id,
    CONCAT('192.168.', FLOOR(RAND() * 255), '.', FLOOR(RAND() * 255)) as ip_address,
    DATE_SUB(NOW(), INTERVAL FLOOR(RAND() * 30) DAY) as date_creation
FROM produits p
CROSS JOIN (
    SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5 
    UNION SELECT 6 UNION SELECT 7 UNION SELECT 8 UNION SELECT 9 UNION SELECT 10
) as numbers
WHERE p.statut = 'approuve'
LIMIT 100;

-- Voir les produits avec leurs vues
SELECT 
    p.id,
    p.nom_article,
    COUNT(pv.id) as nb_vues
FROM produits p
LEFT JOIN produit_views pv ON pv.produit_id = p.id
WHERE p.statut = 'approuve'
GROUP BY p.id
ORDER BY nb_vues DESC
LIMIT 10;
